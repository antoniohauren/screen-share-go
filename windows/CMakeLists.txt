cmake_minimum_required(VERSION 3.24)
project(screen_share_capture LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build using Visual Studio 2022 and Windows 11 SDK")
endif()
add_executable(capture-audio capture-audio/main.cpp)
target_compile_features(capture-audio PRIVATE cxx_std_17)
target_compile_definitions(capture-audio PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(capture-audio PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(capture-audio PRIVATE ole32 mmdevapi user32)
set_property(TARGET capture-audio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
